struct Point { int x, y; };

Point midpoint(Point a, Point b) {
    return { (a.x + b.x) / 2, (a.y + b.y) / 2 };
}

bool operator==(Point a, Point b) {
    return a.x == b.x && a.y == b.y;
}

    Point p = midpoint({0, 0}, {4, 8});
